async function processMessage() { const input = document.getElementById('user-input'); const flow = document.getElementById('chat-flow'); const text = input.value.trim(); if (!text) return; // إضافة رسالة المستخدم flow.innerHTML += `
${text}
`; input.value = ""; // إضافة مؤشر التحميل (Loading Indicator) const loadingId = "bot-" + Date.now(); flow.innerHTML += `
شاهين يحلل طلبك...
`; flow.scrollTop = flow.scrollHeight; try { const res = await fetch('https://shaheen-backend-a5huxtb3zxkpvacbggancq.streamlit.app/chat', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({ message: text }) }); const data = await res.json(); // عرض الرد النهائي بسلاسة document.getElementById(loadingId).innerHTML = `شاهين شات:
${data.reply}`; } catch (e) { // رسالة بسيطة عند الانقطاع المؤقت بدلاً من الرسائل التقنية document.getElementById(loadingId).innerHTML = "نعتذر عن التأخير، يبدو أن هناك ضغطاً لحظياً. يرجى إعادة إرسال الرسالة."; } flow.scrollTop = flow.scrollHeight; }
